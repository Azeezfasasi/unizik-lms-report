1. Project Overview
The UNIZIK Postgraduate LMS Report Portal is a role-based reporting and academic
tracking PWA covering two major flows:
1. Academic Configuration — Faculty/Department structuring, Curriculum management,
Examiner Reports.
2. Student Reporting Pipeline — Faculty → Curriculum → Lectures Calendar →
Examination Timetable → General Reports → Submit → Export (PDF/Excel).

4. MongoDB / Mongoose Data Models
4.1 User.ts
import { Schema, model, models } from "mongoose";
const UserSchema = new Schema(
  {
    name: { type: String, required: true },
    email: { type: String, required: true, unique: true },
    password: { type: String, required: true, select: false },
    role: {
      type: String,
      enum: ["ADMIN", "FACULTY_OFFICER", "EXAMINER", "STUDENT"],
      default: "STUDENT",
    },
    avatarUrl: { type: String },        // Cloudinary secure_url
    facultyId: { type: Schema.Types.ObjectId, ref: "Faculty" },
    isActive: { type: Boolean, default: true },
    lastLoginAt: { type: Date },
    resetToken: { type: String, select: false },
    resetTokenExpiry: { type: Date, select: false },
  },
  { timestamps: true }
);
export default models.User || model("User", UserSchema);

7. Cloudinary Integration
// lib/cloudinary/cloudinary.ts
import { v2 as cloudinary } from "cloudinary";
cloudinary.config({
  cloud_name: process.env.CLOUDINARY_CLOUD_NAME,
  api_key: process.env.CLOUDINARY_API_KEY,
  api_secret: process.env.CLOUDINARY_API_SECRET,
});
export default cloudinary;
// app/api/upload/route.ts
import cloudinary from "@/lib/cloudinary/cloudinary";
export async function POST(req: Request) {
  const { file, folder } = await req.json();
  const result = await cloudinary.uploader.upload(file, {
    folder: folder || "unizik-lms",
    resource_type: "auto",
  });
  return Response.json({ url: result.secure_url, publicId: result.public_id });
}

10. MongoDB Connection Helper
// lib/db/connectDB.ts
import mongoose from "mongoose";
const MONGODB_URI = process.env.MONGODB_URI as string;
let cached = (global as any).mongoose || { conn: null, promise: null };
export async function connectDB() {
  if (cached.conn) return cached.conn;
  if (!cached.promise) {
    cached.promise = mongoose.connect(MONGODB_URI).then((m) => m);
  }
  cached.conn = await cached.promise;
  (global as any).mongoose = cached;
  return cached.conn;
}
